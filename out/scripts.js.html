/**
 * Representa una tarea con un texto descriptivo y su estado de completado.
 *
 * @class Task
 */
class Task {
    /**
     * Crea una nueva tarea.
     *
     * @constructor
     * @param {string} text - El texto descriptivo de la tarea.
     */
    constructor(text) {
        /**
         * Texto descriptivo de la tarea.
         * @type {string}
         */
        this.text = text;
        /**
         * Estado de completado de la tarea.
         * @type {boolean}
         */
        this.completed = false;
    }
}

/**
 * Gestiona una lista de tareas, permitiendo añadir, eliminar, alternar su estado
 * y mantenerlas sincronizadas con el almacenamiento local.
 *
 * @class TaskManager
 */
class TaskManager {
    /**
     * Crea una instancia de TaskManager y carga las tareas desde el almacenamiento local.
     *
     * @constructor
     */
    constructor() {
        /**
         * Lista de tareas.
         * @type {Task[]}
         */
        this.tasks = JSON.parse(localStorage.getItem('tasks')) || [];
    }

    /**
     * Añade una nueva tarea a la lista y actualiza el almacenamiento local.
     *
     * @param {string} text - El texto descriptivo de la nueva tarea.
     */
    addTask(text) {
        const task = new Task(text);
        this.tasks.push(task);
        this.updateLocalStorage();
    }

    /**
     * Elimina la tarea en el índice especificado y actualiza el almacenamiento local.
     *
     * @param {number} index - El índice de la tarea a eliminar.
     */
    removeTask(index) {
        this.tasks.splice(index, 1);
        this.updateLocalStorage();
    }

    /**
     * Alterna el estado de completado de la tarea en el índice especificado y actualiza el almacenamiento local.
     *
     * @param {number} index - El índice de la tarea a modificar.
     */
    toggleTaskCompleted(index) {
        this.tasks[index].completed = !this.tasks[index].completed;
        this.updateLocalStorage();
    }

    /**
     * Actualiza el almacenamiento local con la lista actual de tareas.
     */
    updateLocalStorage() {
        localStorage.setItem('tasks', JSON.stringify(this.tasks));
    }

    /**
     * Obtiene la lista de tareas.
     *
     * @returns {Task[]} La lista de tareas gestionadas.
     */
    getTasks() {
        return this.tasks;
    }
}

/**
 * Instancia del gestor de tareas.
 * @type {TaskManager}
 */
const taskManager = new TaskManager();

/**
 * Añade una tarea utilizando el valor del input y renderiza la lista actualizada de tareas.
 *
 * @function addTask
 */
function addTask() {
    const taskInput = document.getElementById('taskInput');
    const text = taskInput.value.trim();
    if (text) {
        taskManager.addTask(text);
        taskInput.value = '';
        renderTasks();
    }
}

/**
 * Elimina la tarea en el índice especificado y vuelve a renderizar la lista de tareas.
 *
 * @function deleteTask
 * @param {number} index - El índice de la tarea a eliminar.
 */
function deleteTask(index) {
    taskManager.removeTask(index);
    renderTasks();
}

/**
 * Renderiza la lista de tareas en el elemento con id 'taskList'.
 *
 * Por cada tarea, crea un elemento `<li>` que contiene el texto de la tarea y un botón para borrarla.
 * Si la tarea está completada, el texto se muestra tachado.
 *
 * @function renderTasks
 */
function renderTasks() {
    const taskList = document.getElementById('taskList');
    taskList.innerHTML = '';
    taskManager.getTasks().forEach((task, index) => {
        const taskEl = document.createElement('li');
        const taskText = document.createElement('span');
        taskText.textContent = task.text;
        taskText.style.flexGrow = '1';
        if (task.completed) {
            taskText.style.textDecoration = 'line-through';
        }

        // Botón para borrar la tarea
        const deleteBtn = document.createElement('button');
        deleteBtn.textContent = 'Borrar';
        deleteBtn.onclick = () => deleteTask(index);
        deleteBtn.style.marginLeft = '10px';
        deleteBtn.classList.add('buttonB'); // Añadir clase buttonB

        taskEl.appendChild(taskText);
        taskEl.appendChild(deleteBtn);
        taskList.appendChild(taskEl);
    });
}

/**
 * Alterna el estado de completado de la tarea en el índice especificado y vuelve a renderizar la lista de tareas.
 *
 * @function toggleTaskCompleted
 * @param {number} index - El índice de la tarea a alternar.
 */
function toggleTaskCompleted(index) {
    taskManager.toggleTaskCompleted(index);
    renderTasks();
}

// Asigna el evento de clic para añadir una nueva tarea
document.getElementById('addTaskBtn').addEventListener('click', addTask);

// Renderiza las tareas almacenadas al cargar el script
renderTasks();
